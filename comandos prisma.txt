# 1. Cria o arquivo package.json inicial
npm init -y

# 2. Instala as dependências principais da aplicação
npm install express @prisma/client dotenv

# 3. Instala as ferramentas de desenvolvimento (Prisma CLI e Nodemon para rodar o servidor)
npm install -D prisma nodemon

# 4. Inicializa a estrutura do Prisma (cria a pasta /prisma com o schema.prisma e o arquivo .env)
npx prisma init

#5. No .env
DATABASE_URL="postgresql://postgres:SUA_SENHA@localhost:5432/trabalho_web?schema=public"

# Executa a migração: cria o banco de dados, gera as tabelas no PostgreSQL e constrói o Prisma Client
npx prisma migrate dev --name cria_alunos_e_professores

# Abre o painel visual do Prisma Studio no navegador (http://localhost:5555)
npx prisma studio
